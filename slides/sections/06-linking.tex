% --------- Slide 13: Dereferenceable IRIs ---------
\begin{frame}[fragile]{Dereferenceable IRIs: 303 and Content Negotiation}
    \footnotesize
    \begin{tabularx}{\textwidth}{@{}l>{\raggedright\arraybackslash}X@{}}
        \toprule
        Route & Response \\
        \midrule
        \term{/resource/\{name\}} & 303 $\rightarrow$ HTML page, or $\rightarrow$
            \term{/data/\{name\}.ext} when \term{Accept} asks for RDF \\
        \term{/page/\{name\}}     & 303 $\rightarrow$ HTML page, like \term{dbpedia.org/page/…} \\
        \term{/data/\{name\}.ext} & triples about the resource: \term{.ttl}, \term{.nt}, \term{.jsonld}, \term{.rdf} \\
        \term{/ontology/\{term\}} & definition of a \term{vio:} term, in Turtle \\
        \bottomrule
    \end{tabularx}

    \vspace{0.6em}
    Recorded on the running server:
\begin{Verbatim}[fontsize=\tiny]
$ curl -si -H "Accept: text/turtle" \
       http://127.0.0.1:7860/resource/Nguy%E1%BB%85n_C%C3%B4ng_Ph%C6%B0%E1%BB%A3ng
HTTP/1.1 303 See Other
vary: Accept
location: /data/Nguy%E1%BB%85n_C%C3%B4ng_Ph%C6%B0%E1%BB%A3ng.ttl

$ curl -sL -H "Accept: text/turtle" http://127.0.0.1:7860/resource/Nguy%E1%BB%85n_…
vres:Nguyễn_Công_Phượng a dbo:Animal, dbo:Athlete, dbo:Person, dbo:SoccerPlayer, … ;
    dbo:birthDate "1995-01-21"^^xsd:date ;
    …
    vio:birthProvince vres:Nghệ_An ;
    …
    owl:sameAs dbr:Nguyễn_Công_Phượng, wd:Q18045362 ;
\end{Verbatim}
    \scriptsize
    Headers and Turtle abridged. Browsers get the resource page at \term{/?resource=<name>}, a shareable URL.
    Clients must percent-encode the Vietnamese characters, as browsers do.
\end{frame}
