% --------- Slide 18: Limitations and future work ---------
\begin{frame}{Limitations and Future Work}
    \begin{block}{Limitations}
        \footnotesize
        \begin{itemize}
            \item One domain, 990 entities --- each new class needs a seed query,
                  ontology terms, a mapping and a build function
            \item 44 of 777 DBpedia links are stale (12 redirects, 32 missing)
            \item 129 career stations without a team IRI; 26 universities without a province
            \item No SPARQL 1.1 Protocol endpoint (\term{/sparql}), no query time limit,
                  \term{SERVICE} not blocked
            \item Question answering not measured on a question benchmark
        \end{itemize}
    \end{block}
    \begin{block}{Future work}
        \footnotesize
        \begin{itemize}
            \item \term{/sparql} endpoint with time limits; follow DBpedia redirects at build time
            \item More classes (coaches, competitions) and page links (\term{dbo:wikiPageWikiLink})
            \item A benchmark of Vietnamese questions for the question-answering tab
        \end{itemize}
    \end{block}
\end{frame}
